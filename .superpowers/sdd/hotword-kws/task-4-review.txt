package ru.samokontrol.hotwordspike.kws

import android.content.res.AssetManager
import java.io.File
import java.io.InputStream
import java.nio.ByteBuffer
import java.nio.ByteOrder

/**
 * Хранилище эталонов и порога.
 *
 * `templates.bin`: ASCII `HWK1`, затем `int count`, `int rows`, `int dims`,
 * далее `count * rows * dims` значений `float` little-endian.
 *
 * Если шаблоны имеют разное число кадров (как в round-trip тесте), поле `rows`
 * равно 0, а перед данными каждого шаблона пишется его собственный `int rows`.
 * Такой файл читается только этим загрузчиком; прямоугольные файлы полностью
 * соответствуют формату выше.
 *
 * `threshold.txt`: одна строка с числом (`Float.toString`).
 */
object TemplateStore {
    private val MAGIC = byteArrayOf('H'.code.toByte(), 'W'.code.toByte(), 'K'.code.toByte(), '1'.code.toByte())
    private const val HEADER_BYTES = 4 + 4 + 4 + 4
    const val TEMPLATES_FILE = "templates.bin"
    const val THRESHOLD_FILE = "threshold.txt"

    fun saveTemplates(dir: File, name: String, templates: List<Array<FloatArray>>) {
        val dims = templates.firstOrNull()?.firstOrNull()?.size ?: 0
        templates.forEach { template ->
            template.forEach { frame ->
                require(frame.size == dims) { "Все кадры должны иметь одинаковую размерность" }
            }
        }
        val uniformRows = templates.all { it.size == (templates.firstOrNull()?.size ?: 0) }
        val rows = if (uniformRows) templates.firstOrNull()?.size ?: 0 else 0

        var size = HEADER_BYTES + templates.size * dims * 4
        templates.forEach { template ->
            size += template.size * dims * 4
            if (!uniformRows) size += 4
        }

        val buffer = ByteBuffer.allocate(size).order(ByteOrder.LITTLE_ENDIAN)
        buffer.put(MAGIC)
        buffer.putInt(templates.size)
        buffer.putInt(rows)
        buffer.putInt(dims)
        templates.forEach { template ->
            if (!uniformRows) buffer.putInt(template.size)
            template.forEach { frame ->
                frame.forEach { value -> buffer.putFloat(value) }
            }
        }

        dir.mkdirs()
        File(dir, name).writeBytes(buffer.array())
    }

    fun loadTemplatesFromDir(dir: File, name: String): List<Array<FloatArray>> {
        val file = File(dir, name)
        if (!file.exists()) return emptyList()
        return file.inputStream().use { readTemplates(it) }
    }

    fun loadTemplates(assets: AssetManager, path: String): List<Array<FloatArray>> =
        assets.open(path).use { readTemplates(it) }

    fun saveThreshold(dir: File, value: Float) {
        dir.mkdirs()
        File(dir, THRESHOLD_FILE).writeText(value.toString())
    }

    fun loadThreshold(assets: AssetManager, path: String): Float =
        assets.open(path).use { it.readThreshold() }

    fun loadThresholdFromDir(dir: File): Float =
        File(dir, THRESHOLD_FILE).inputStream().use { it.readThreshold() }

    private fun readTemplates(input: InputStream): List<Array<FloatArray>> {
        val bytes = input.readBytes()
        if (bytes.size < HEADER_BYTES) return emptyList()
        val buffer = ByteBuffer.wrap(bytes).order(ByteOrder.LITTLE_ENDIAN)
        val magic = ByteArray(4).also { buffer.get(it) }
        if (!magic.contentEquals(MAGIC)) return emptyList()

        val count = buffer.int
        val headerRows = buffer.int
        val dims = buffer.int
        if (count < 0 || headerRows < 0 || dims < 0) return emptyList()

        val templates = ArrayList<Array<FloatArray>>(count)
        repeat(count) {
            val rows = if (headerRows > 0) headerRows else buffer.int
            if (rows < 0 || buffer.remaining() < rows * dims * 4) return emptyList()
            templates += Array(rows) { FloatArray(dims) { buffer.float } }
        }
        return templates
    }

    private fun InputStream.readThreshold(): Float =
        bufferedReader().use { it.readLine() }.trim().toFloat()
}

package ru.samokontrol.hotwordspike.kws

import android.media.AudioFormat
import android.media.AudioRecord
import android.media.MediaRecorder
import kotlin.math.max
import kotlin.math.sqrt

/**
 * Запись образцов с микрофона: 16 кГц, моно, PCM16, [MediaRecorder.AudioSource.VOICE_RECOGNITION].
 *
 * Аудио в файлы не пишется: [record] возвращает мел-признаки, [recordRaw] — сырые сэмплы
 * для калибровки. [AudioRecord] всегда освобождается в `finally`.
 */
class SampleRecorder(
    private val features: MelFeatures,
    private val sampleRate: Int = 16000,
) {
    /**
     * Блокирующая запись [durationMs], обрезка тишины по краям.
     *
     * @throws IllegalStateException если [AudioRecord] не инициализировался.
     */
    fun record(durationMs: Int = 2200): Array<FloatArray> {
        val raw = recordRaw(durationMs)
        return features.logMel(trimSilence(raw))
    }

    /**
     * Запись фона [durationMs] без обрезки — для калибровки нужен сырой поток.
     *
     * @throws IllegalStateException если [AudioRecord] не инициализировался.
     */
    fun recordRaw(durationMs: Int = 30_000): ShortArray {
        val totalSamples = max(1, sampleRate * durationMs / 1000)
        val minBuffer = AudioRecord.getMinBufferSize(
            sampleRate,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT,
        )
        if (minBuffer <= 0) throw IllegalStateException("AudioRecord: неверные параметры записи")

        val record = AudioRecord(
            MediaRecorder.AudioSource.VOICE_RECOGNITION,
            sampleRate,
            AudioFormat.CHANNEL_IN_MONO,
            AudioFormat.ENCODING_PCM_16BIT,
            max(minBuffer, sampleRate),
        )
        try {
            if (record.state != AudioRecord.STATE_INITIALIZED) {
                throw IllegalStateException("AudioRecord не инициализировался (state=${record.state})")
            }
            val result = ShortArray(totalSamples)
            record.startRecording()
            var read = 0
            while (read < totalSamples) {
                val chunk = record.read(result, read, totalSamples - read)
                if (chunk <= 0) break
                read += chunk
            }
            return if (read == totalSamples) result else result.copyOf(read)
        } finally {
            try {
                record.stop()
            } catch (_: IllegalStateException) {
            }
            record.release()
        }
    }

    /**
     * Обрезка тишины: RMS по кадрам 20 мс, порог — 10% от максимального RMS.
     * Оставляем не меньше 1 с, расширяя область симметрично вокруг центра.
     */
    private fun trimSilence(samples: ShortArray): ShortArray {
        val frame = max(1, sampleRate / 50)
        if (samples.size <= frame) return samples

        val frameCount = samples.size / frame
        val rms = DoubleArray(frameCount)
        var maxRms = 0.0
        for (i in 0 until frameCount) {
            var sum = 0.0
            val offset = i * frame
            for (j in offset until offset + frame) {
                val value = samples[j].toDouble()
                sum += value * value
            }
            rms[i] = sqrt(sum / frame)
            if (rms[i] > maxRms) maxRms = rms[i]
        }

        val gate = maxRms * 0.1
        var first = 0
        while (first < frameCount && rms[first] < gate) first++
        var last = frameCount - 1
        while (last >= first && rms[last] < gate) last--

        if (first > last) {
            // Полная тишина: берём центральную секунду, чтобы не отдавать пустой массив.
            val start = ((samples.size - sampleRate) / 2).coerceIn(0, samples.size)
            return samples.copyOfRange(start, minOf(samples.size, start + sampleRate))
        }

        var start = first * frame
        var end = minOf(samples.size, (last + 1) * frame)
        val minLength = minOf(sampleRate, samples.size)
        if (end - start < minLength) {
            val center = (start + end) / 2
            start = (center - minLength / 2).coerceIn(0, samples.size - minLength)
            end = start + minLength
        }
        return samples.copyOfRange(start, end)
    }
}

class MainActivity : Activity() {

    private fun requestAndStart() {
        val missing = mutableListOf<String>()
        if (!hasAudioPermission()) {
            missing += Manifest.permission.RECORD_AUDIO
        }
        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.TIRAMISU &&
            checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) {
            missing += Manifest.permission.POST_NOTIFICATIONS
        }
        if (missing.isEmpty()) {
            start()
        } else {
            requestPermissions(missing.toTypedArray(), REQUEST_PERMISSIONS)
        }
    }

    override fun onRequestPermissionsResult(
        requestCode: Int,
        permissions: Array<out String>,
        grantResults: IntArray,
    ) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults)
        when (requestCode) {
            REQUEST_PERMISSIONS -> {
                if (grantResults.all { it == PackageManager.PERMISSION_GRANTED }) {
                    start()
                } else {
                    status.text = "Без разрешений сервис не запустится."
                }
            }

            REQUEST_AUDIO -> {
                val action = pendingAction
                pendingAction = null
                if (hasAudioPermission()) {
                    action?.invoke()
                } else {
                    setStatus("Без доступа к микрофону запись невозможна.")
                }
            }
        }
    }

    private fun start() {
        try {
            ListenService.start(this)
            status.text = "Сервис запущен."
        } catch (t: Throwable) {
            status.text = "Не удалось запустить: ${t.javaClass.simpleName} ${t.message}"
        }
        refresh()
    }

    private fun refresh() {
        val file = File(createDeviceProtectedStorageContext().filesDir, "spike.log")
        val tail = if (file.exists()) {
            file.readLines().takeLast(60).joinToString("\n")
        } else {
            "Лог пуст."
        }
        status.text = "Лог: ${file.absolutePath}\n\n$tail"
    }

    private fun hasAudioPermission(): Boolean =
        checkSelfPermission(Manifest.permission.RECORD_AUDIO) == PackageManager.PERMISSION_GRANTED

    private fun withAudioPermission(action: () -> Unit) {
        if (hasAudioPermission()) {
            action()
        } else {
            pendingAction = action
            requestPermissions(arrayOf(Manifest.permission.RECORD_AUDIO), REQUEST_AUDIO)
        }
    }

    private fun recordSamples() {
        runInBackground {
            setStatus("Запись образцов: приготовьтесь…")
            val recorded = ArrayList<Array<FloatArray>>(SAMPLE_COUNT)
            for (index in 1..SAMPLE_COUNT) {
                setStatus("Образец $index/$SAMPLE_COUNT")
                recorded += recorder.record(SAMPLE_MILLIS)
                if (index < SAMPLE_COUNT) Thread.sleep(PAUSE_MILLIS)
            }
            templates = recorded
            logSelfScores(recorded)
            setStatus("Готово: $SAMPLE_COUNT образцов")
            logger.log("samples recorded count=$SAMPLE_COUNT")
        }
    }

    private fun logSelfScores(samples: List<Array<FloatArray>>) {
        val scores = samples.mapIndexed { index, sample ->
            var best = Float.MAX_VALUE
            samples.forEachIndexed { other, candidate ->
                if (other != index) {
                    val score = Dtw.score(candidate, sample)
                    if (score < best) best = score
                }
            }
            logger.log("sample ${index + 1} self=$best")
            best
        }
        selfMin = scores.minOrNull() ?: 0f
        selfMean = if (scores.isEmpty()) 0f else scores.average().toFloat()
        logger.log("selfScores min=$selfMin mean=$selfMean")
    }

    private fun recordBackground() {
        if (templates.isEmpty()) {
            setStatus("Сначала запишите образцы.")
            return
        }
        runInBackground {
            if (selfMin == 0f && templates.size > 1) logSelfScores(templates)
            setStatus("Запись фона 30 с…")
            val raw = recorder.recordRaw(BACKGROUND_MILLIS)
            setStatus("Считаю порог…")
            val features = melFeatures.logMel(raw)
            val minBackground = minBackgroundScore(features)
            if (minBackground == Float.MAX_VALUE) {
                setStatus("Недостаточно фона для калибровки (${features.size} кадров).")
                return@runInBackground
            }
            val calibrated = (minBackground - THRESHOLD_MARGIN).coerceIn(MIN_THRESHOLD, MAX_THRESHOLD)
            threshold = calibrated
            TemplateStore.saveThreshold(hotwordDir, calibrated)
            val message = "calibration selfMin=$selfMin selfMean=$selfMean bgMin=$minBackground threshold=$calibrated"
            logger.log(message)
            setStatus("Порог: $calibrated\n$message")
        }
    }

    private fun minBackgroundScore(features: Array<FloatArray>): Float {
        if (features.size < WINDOW_FRAMES) return Float.MAX_VALUE
        var minBackground = Float.MAX_VALUE
        var start = 0
        while (start + WINDOW_FRAMES <= features.size) {
            val window = features.copyOfRange(start, start + WINDOW_FRAMES)
            var best = Float.MAX_VALUE
            templates.forEach { template ->
                val score = Dtw.score(template, window)
                if (score < best) best = score
            }
            if (best < minBackground) minBackground = best
            start += WINDOW_STEP
        }
        return minBackground
    }

    private fun buildTemplates() {
        if (templates.isEmpty()) {
            setStatus("Сначала запишите образцы.")
            return
        }
        val value = threshold
        if (value == null) {
            setStatus("Сначала запишите фон 30 с для калибровки.")
            return
        }
        runInBackground {
            TemplateStore.saveTemplates(hotwordDir, TemplateStore.TEMPLATES_FILE, templates)
            TemplateStore.saveThreshold(hotwordDir, value)
            setStatus(
                "Сохранено: ${File(hotwordDir, TemplateStore.TEMPLATES_FILE).absolutePath}\n" +
                    "Порог: ${File(hotwordDir, TemplateStore.THRESHOLD_FILE).absolutePath}",
            )
            logger.log("templates saved count=${templates.size} threshold=$value")
        }
    }

    private fun exportFiles() {
        runInBackground {
            val target = getExternalFilesDir(null)
            if (target == null) {
                setStatus("Внешнее хранилище недоступно.")
                return@runInBackground
            }
            target.mkdirs()
            val names = listOf(TemplateStore.TEMPLATES_FILE, TemplateStore.THRESHOLD_FILE)
            val copied = names.mapNotNull { name ->
                val source = File(hotwordDir, name)
                if (!source.exists()) return@mapNotNull null
                val dest = File(target, name)
                source.copyTo(dest, overwrite = true)
                dest
            }
            if (copied.isEmpty()) {
                setStatus("Нечего выгружать: сначала соберите файлы.")
            } else {
                setStatus("Выгружено:\n" + copied.joinToString("\n") { it.absolutePath })
                logger.log("exported ${copied.joinToString { it.absolutePath }}")
            }
        }
    }

    private fun loadSavedCalibration() {
        val savedTemplates = TemplateStore.loadTemplatesFromDir(hotwordDir, TemplateStore.TEMPLATES_FILE)
        if (savedTemplates.isNotEmpty()) templates = savedTemplates
        threshold = runCatching { TemplateStore.loadThresholdFromDir(hotwordDir) }.getOrNull()
        logger.log("loaded templates=${templates.size} threshold=${threshold ?: "none"}")
    }

    private fun runInBackground(block: () -> Unit) {
        if (busy) {
            setStatus("Дождитесь завершения текущей операции.")
            return
        }
        busy = true
        Thread {
            try {
                block()
            } catch (t: Throwable) {
                logger.log("operation failed: ${t.javaClass.simpleName} ${t.message}")
                setStatus("Ошибка: ${t.javaClass.simpleName} ${t.message}")
            } finally {
                busy = false
            }
        }.start()
    }

    private fun setStatus(text: String) {
        runOnUiThread {
            if (!isFinishing && !isDestroyed) status.text = text
        }
    }

    companion object {
        private const val REQUEST_PERMISSIONS = 1
        private const val REQUEST_AUDIO = 2

        private const val SAMPLE_COUNT = 10
        private const val SAMPLE_MILLIS = 2200
        private const val PAUSE_MILLIS = 700L
        private const val BACKGROUND_MILLIS = 30_000

        private const val WINDOW_FRAMES = 200
        private const val WINDOW_STEP = 10
        private const val THRESHOLD_MARGIN = 0.05f
        private const val MIN_THRESHOLD = 0.01f
        private const val MAX_THRESHOLD = 5f
    }
}

package ru.samokontrol.hotwordspike.kws

import java.nio.file.Files
import org.junit.Assert.assertEquals
import org.junit.Test

class TemplateStoreTest {
    @Test
    fun `templates survive round trip`() {
        val dir = Files.createTempDirectory("hwk").toFile()
        val t = listOf(Array(3) { floatArrayOf(1f, -2.5f) }, Array(2) { floatArrayOf(0.25f, 7f) })
        TemplateStore.saveTemplates(dir, "t.bin", t)
        val back = TemplateStore.loadTemplatesFromDir(dir, "t.bin")
        assertEquals(2, back.size)
        assertEquals(3, back[0].size)
        assertEquals(-2.5f, back[0][1][1], 0.0001f)
        assertEquals(7f, back[1][1][1], 0.0001f)
    }

    @Test
    fun `threshold survives round trip`() {
        val dir = Files.createTempDirectory("hwk").toFile()
        TemplateStore.saveThreshold(dir, 0.42f)
        assertEquals(0.42f, TemplateStore.loadThresholdFromDir(dir), 0.0001f)
    }

    @Test
    fun `templates file uses little endian`() {
        val dir = Files.createTempDirectory("hwk").toFile()
        TemplateStore.saveTemplates(dir, "t.bin", listOf(Array(1) { floatArrayOf(1f) }))
        val bytes = java.io.File(dir, "t.bin").readBytes()
        assertEquals('H'.code.toByte(), bytes[0])
        assertEquals('W'.code.toByte(), bytes[1])
        assertEquals('K'.code.toByte(), bytes[2])
        assertEquals('1'.code.toByte(), bytes[3])
        // count = 1 (little-endian)
        assertEquals(1.toByte(), bytes[4])
        assertEquals(0.toByte(), bytes[5])
        // float 1.0f little-endian = 00 00 80 3F
        assertEquals(0.toByte(), bytes[16])
        assertEquals(0.toByte(), bytes[17])
        assertEquals(0x80.toByte(), bytes[18])
        assertEquals(0x3F.toByte(), bytes[19])
    }
}
